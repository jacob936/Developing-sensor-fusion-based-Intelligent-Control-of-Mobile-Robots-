import pandas as pd, numpy as np, matplotlib.pyplot as plt, glob

# Load data
asmc = pd.read_csv(sorted(glob.glob('/root/sensor_fusion_ws/experiment_data/asmc_tune_*.csv'))[-1])
pid = pd.read_csv(sorted(glob.glob('/root/sensor_fusion_ws/experiment_data/pid_tune_*.csv'))[-1])

# Side-by-side comparison
fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(14, 10))

# Trajectory comparison
ax1.plot(asmc['ref_x'], asmc['ref_y'], 'g--', label='Waypoints', alpha=0.5)
ax1.plot(asmc['state_x'], asmc['state_y'], 'b-', label='ASMC', linewidth=1.5)
ax1.set_title('ASMC Trajectory')
ax1.legend(); ax1.grid(True, alpha=0.3); ax1.axis('equal')

ax2.plot(pid['ref_x'], pid['ref_y'], 'g--', label='Waypoints', alpha=0.5)
ax2.plot(pid['state_x'], pid['state_y'], 'r-', label='PID', linewidth=1.5)
ax2.set_title('PID Trajectory')
ax2.legend(); ax2.grid(True, alpha=0.3); ax2.axis('equal')

# Error comparison
ax3.plot(asmc['time_sec'], asmc['error_dist'], 'b-', label='ASMC Error', alpha=0.7)
ax3.set_title('Tracking Error')
ax3.set_ylabel('Error (m)')
ax3.legend(); ax3.grid(True, alpha=0.3)

ax4.plot(pid['time_sec'], pid['error_dist'], 'r-', label='PID Error', alpha=0.7)
ax4.set_title('Tracking Error')
ax4.set_xlabel('Time (s)')
ax4.set_ylabel('Error (m)')
ax4.legend(); ax4.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('/root/sensor_fusion_ws/experiment_data/comparison_trajectories.png', dpi=300)
print("✓ Saved: comparison_trajectories.png")
plt.show()
